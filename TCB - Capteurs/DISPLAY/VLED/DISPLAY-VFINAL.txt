#include "led-matrix.h"
#include "graphics.h"
#include <mosquitto.h>
#include <unistd.h>
#include <signal.h>
#include <iostream>
#include <string>
#include <sstream>
#include <jsoncpp/json/json.h>

using namespace rgb_matrix;

volatile bool interrupt_received = false;
RGBMatrix *matrix = nullptr;
FrameCanvas *canvas = nullptr;
Font font;

static void InterruptHandler(int signo) {
    interrupt_received = true;
}

// Message "BROKER OFFLINE"
void FlashOfflineMessage(FrameCanvas *canvas, Font &font) {
    Color red(255, 0, 0);
    Color black(0, 0, 0);
    for (int i = 0; i < 6 && !interrupt_received; ++i) {
        canvas->Clear();
        if (i % 2 == 0) {
            DrawText(canvas, font, 5, 30, red, nullptr, "BROKER OFFLINE", 0);
        }
        canvas = matrix->SwapOnVSync(canvas);
        sleep(1);
    }
}

// Affiche les scores
static void DrawScore(FrameCanvas *canvas, const Json::Value &data) {
    canvas->Clear();
    Color jaune(255, 255, 0);
    Color rouge(255, 0, 0);

    int y = 5;

    // En-tête
    DrawText(canvas, font, 13, y, jaune, nullptr, "SET", 0);
    DrawText(canvas, font, 30, y, jaune, nullptr, "JEUX", 0);
    DrawText(canvas, font, 50, y, jaune, nullptr, "PTS", 0);

    // JOUEUR1
    y += 7;
    DrawText(canvas, font, 0, y, jaune, nullptr, "J1", 0);
    DrawText(canvas, font, 18, y, jaune, nullptr, std::to_string(data["home_sets"].asInt()).c_str(), 0);
    DrawText(canvas, font, 35, y, jaune, nullptr, std::to_string(data["home_points"].asInt()).c_str(), 0);
    DrawText(canvas, font, 50, y, jaune, nullptr, std::to_string(data["home_total"].asInt()).c_str(), 0);

    // Séparateur
    y += 6;
    for (int x = 0; x < 64; x += 4)
        DrawText(canvas, font, x, y, jaune, nullptr, "-", 0);

    // JOUEUR2
    y += 6;
    DrawText(canvas, font, 0, y, jaune, nullptr, "J2", 0);
    DrawText(canvas, font, 18, y, jaune, nullptr, std::to_string(data["visit_sets"].asInt()).c_str(), 0);
    DrawText(canvas, font, 35, y, jaune, nullptr, std::to_string(data["visit_points"].asInt()).c_str(), 0);
    DrawText(canvas, font, 50, y, jaune, nullptr, std::to_string(data["visit_total"].asInt()).c_str(), 0);

    // previous set
    y += 7;
    DrawText(canvas, font, 10, y, jaune, nullptr, "previous set", 0);

    y += 7;
    DrawText(canvas, font, 0, y, jaune, nullptr, "J1", 0);
    DrawText(canvas, font, 20, y, jaune, nullptr, std::to_string(data["home_prev_games"].asInt()).c_str(), 0);

    y += 6;
    DrawText(canvas, font, 0, y, jaune, nullptr, "J2", 0);
    DrawText(canvas, font, 20, y, jaune, nullptr, std::to_string(data["visit_prev_games"].asInt()).c_str(), 0);

    // ligne séparatrice
    y += 6;
    for (int x = 0; x < 64; x += 4)
        DrawText(canvas, font, x, y, jaune, nullptr, "-", 0);

    // Message LET si détecté
    y += 7;
    if (data["let_detected"].asBool()) {
        DrawText(canvas, font, 10, y, rouge, nullptr, "Message LET", 0);
    }

    canvas = matrix->SwapOnVSync(canvas);
}



// Callback de message
void mqtt_callback(struct mosquitto *mosq, void *obj, const struct mosquitto_message *msg) {
    if (!msg->payload) return;

    std::string payload(static_cast<char *>(msg->payload));
    Json::Value root;
    Json::CharReaderBuilder builder;
    std::string errs;

    std::istringstream s(payload);
    if (Json::parseFromStream(builder, s, &root, &errs)) {
        DrawScore(canvas, root);
    } else {
        std::cerr << "Erreur de parsing JSON : " << errs << std::endl;
    }
}

// Callback de reconnexion automatique
void on_disconnect(struct mosquitto *mosq, void *obj, int rc) {
    if (rc != 0) {
        std::cerr << "Déconnexion du broker. Tentative de reconnexion..." << std::endl;
        while (!interrupt_received && mosquitto_reconnect(mosq) != MOSQ_ERR_SUCCESS) {
            FlashOfflineMessage(canvas, font);
        }
        mosquitto_subscribe(mosq, nullptr, "SCORE/TENNIS", 0);
    }
}

int main(int argc, char *argv[]) {
    // Config matrice LED
    RGBMatrix::Options options;
    RuntimeOptions runtime_opt;
    options.rows = 64;
    options.cols = 64;
    options.chain_length = 1;
    options.parallel = 1;
    options.hardware_mapping = "regular";

    matrix = CreateMatrixFromFlags(&argc, &argv, &options, &runtime_opt);
    if (!matrix) return 1;

    canvas = matrix->CreateFrameCanvas();

    if (!font.LoadFont("fonts/4x6.bdf")) {
        std::cerr << "Erreur de chargement de la police 4x6.bdf" << std::endl;
        return 1;
    }

    // Test local initial
    Json::Value test;
    test["home_sets"] = 1;
    test["visit_sets"] = 2;
    test["home_points"] = 2;
    test["visit_points"] = 3;
    test["home_total"] = 15;
    test["visit_total"] = 25;
    test["home_prev_games"] = 6;
    test["home_prev_points"] = 40;
    test["visit_prev_games"] = 4;
    test["visit_prev_points"] = 30;
    DrawScore(canvas, test);
    sleep(3);

    // Init MQTT
    mosquitto_lib_init();
    struct mosquitto *mosq = mosquitto_new("score-tennis", true, nullptr);
    if (!mosq) return 1;

    mosquitto_username_pw_set(mosq, "broker", "tcbBROKER");
    mosquitto_message_callback_set(mosq, mqtt_callback);
    mosquitto_disconnect_callback_set(mosq, on_disconnect);

    // Connexion initiale avec affichage clignotant
    int mqtt_ret = mosquitto_connect(mosq, "10.160.99.33", 1883, 60);
    while (!interrupt_received && mqtt_ret != MOSQ_ERR_SUCCESS) {
        FlashOfflineMessage(canvas, font);
        mqtt_ret = mosquitto_connect(mosq, "10.160.99.33", 1883, 60);
    }

    mosquitto_subscribe(mosq, nullptr, "SCORE/TENNIS", 0);

    signal(SIGTERM, InterruptHandler);
    signal(SIGINT, InterruptHandler);

    while (!interrupt_received) {
        mosquitto_loop(mosq, -1, 1);
    }

    // Clean exit
    mosquitto_destroy(mosq);
    mosquitto_lib_cleanup();
    matrix->Clear();
    delete matrix;

    return 0;
}
